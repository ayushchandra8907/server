30std::string trim_copy(std::string value)
31{
32 auto not_space = [](unsigned char ch) { return !std::isspace(ch); };
33 value.erase(value.begin(),
34 std::find_if(value.begin(), value.end(), not_space));
35 value.erase(std::find_if(value.rbegin(), value.rend(), not_space).base(),
36 value.end());
37 return value;
38}

118std::string duckdb_s3_endpoint_setting(const std::string &endpoint)
119{
120 auto trimmed = trim_copy(endpoint);
121 const std::string https_prefix = "https://";
122 const std::string http_prefix = "http://";
123
124 if (trimmed.rfind(https_prefix, 0) == 0)
125 trimmed.erase(0, https_prefix.size());
126 else if (trimmed.rfind(http_prefix, 0) == 0)
127 trimmed.erase(0, http_prefix.size());
128
129 while (!trimmed.empty() && trimmed.back() == '/')
130 trimmed.pop_back();
131
132 return trimmed;
133}

227bool configure_duckdb_s3(duckdb::Connection *con,
228 const parquet::ObjectStoreConfig &config,
229 std::string *error)
230{
231 if (con == nullptr) {
232 if (error != nullptr)
233 *error = "DuckDB connection must not be null";
234 return false;
235 }
236
237 if (config.bucket.empty()) {
238 if (error != nullptr)
239 *error = "object store bucket is missing";
240 return false;
241 }
242
243 if (config.auth_mode == parquet::ObjectStoreAuthMode::kRemoteSigning) {
244 if (error != nullptr) {
245 *error =
246 "object store auth_mode=remote_signing is not supported by the "
247 "current Parquet handler path";
248 }
249 return false;
250 }
251
252 if (config.credentials.empty()) {
253 if (error != nullptr) {
254 *error =
255 "object store credentials are missing; configure "
256 "parquet_s3_access_key_id and parquet_s3_secret_access_key";
257 }
258 return false;
259 }
260
261 if (config.auth_mode == parquet::ObjectStoreAuthMode::kTemporaryCredentials &&
262 config.credentials.session_token.empty()) {
263 if (error != nullptr) {
264 *error =
265 "object store auth_mode=temporary requires a session_token, which "
266 "is not currently available through Parquet plugin variables";
267 }
268 return false;
269 }
270
271 parquet_log_info("DuckDB configuring S3 endpoint='" + config.endpoint +
272 "' region='" + config.region + "' bucket='" +
273 config.bucket + "' key_prefix='" + config.key_prefix +
274 "' url_style='" + config.url_style + "'");
275
276 if (!config.region.empty()) {
277 con->Query("SET s3_region=" + quote_string_literal(config.region));
278 }
279 con->Query("SET s3_access_key_id=" +
280 quote_string_literal(config.credentials.access_key_id));
281 con->Query("SET s3_secret_access_key=" +
282 quote_string_literal(config.credentials.secret_access_key));
283 if (!config.credentials.session_token.empty()) {
284 con->Query("SET s3_session_token=" +
285 quote_string_literal(config.credentials.session_token));
286 }
287
288 if (!config.endpoint.empty()) {
289 con->Query("SET s3_endpoint=" +
290 quote_string_literal(duckdb_s3_endpoint_setting(config.endpoint)));
291 con->Query(std::string("SET s3_use_ssl=") +
292 (duckdb_s3_use_ssl(config.endpoint) ? "true" : "false"));
293 }
294 con->Query("SET s3_url_style=" +
295 quote_string_literal(duckdb_s3_url_style(config.url_style)));
296 return true;
297}

341std::vector<std::string> extract_manifest_paths(const std::string &response_body)
342{
343 std::vector<std::string> s3_files;
344 std::unordered_set<std::string> seen_s3_files;
345 size_t pos = 0;
346 while ((pos = response_body.find("\"manifest-list\"", pos)) !=
347 std::string::npos) {
348 size_t colon = response_body.find(':', pos);
349 if (colon == std::string::npos)
350 break;
351 size_t value_start = response_body.find('"', colon + 1);
352 if (value_start == std::string::npos)
353 break;
354 size_t value_end = response_body.find('"', value_start + 1);
355 if (value_end == std::string::npos)
356 break;
357
358 std::string path = response_body.substr(value_start + 1,
359 value_end - value_start - 1);
360 if (path.rfind("s3://", 0) == 0 && seen_s3_files.insert(path).second)
361 s3_files.push_back(path);
362
363 pos = value_end + 1;
364 }
365 return s3_files;
366}

368bool fetch_lakekeeper_table_metadata(const parquet::TableMetadata &metadata,
369 std::string *response_body,
370 long *http_code)
371{
372 if (!response_body || !http_code)
373 return false;
374
375 response_body->clear();
376 *http_code = 0;
377
378 const std::string lakekeeper_url = lakekeeper_table_url(metadata);
379 if (lakekeeper_url.empty())
380 return false;
381
382 parquet_log_info("LakeKeeper load table metadata table='" +
383 metadata.catalog_table_ident.table_name + "' url='" +
384 lakekeeper_url + "'");
385
386 CURL *curl = curl_easy_init();
387 if (!curl)
388 return false;
389
390 struct curl_slist *headers = nullptr;
391 apply_catalog_auth_header(metadata.catalog_config, &headers);
392 curl_easy_setopt(curl, CURLOPT_URL, lakekeeper_url.c_str());
393 curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, +curl_write_to_string);
394 curl_easy_setopt(curl, CURLOPT_WRITEDATA, response_body);
395 apply_catalog_curl_options(curl, metadata.catalog_config);
396 if (headers != nullptr)
397 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers);
398
399 CURLcode res = curl_easy_perform(curl);
400 curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, http_code);
401 if (headers != nullptr)
402 curl_slist_free_all(headers);
403 curl_easy_cleanup(curl);
404
405 if (res == CURLE_OK) {
406 parquet_log_info("LakeKeeper load table metadata complete table='" +
407 metadata.catalog_table_ident.table_name +
408 "' http_status=" + std::to_string(*http_code) +
409 " response=" + parquet_log_preview(*response_body));
410 } else {
411 parquet_log_warning("LakeKeeper load table metadata failed table='" +
412 metadata.catalog_table_ident.table_name + "' error='" +
413 std::string(curl_easy_strerror(res)) + "'");
414 }
415
416 return res == CURLE_OK;
417}

419bool resolve_parquet_data_files(const parquet::TableMetadata &metadata,
420 std::vector<std::string> *s3_files,
421 long *http_code)
422{
423 if (!s3_files)
424 return false;
425
426 std::string response_body;
427 long local_http_code = 0;
428 if (!fetch_lakekeeper_table_metadata(metadata, &response_body,
429 &local_http_code)) {
430 if (http_code)
431 *http_code = local_http_code;
432 return false;
433 }
434
435 if (http_code)
436 *http_code = local_http_code;
437
438 if (local_http_code != 200) {
439 s3_files->clear();
440 return true;
441 }
442
443 *s3_files = extract_manifest_paths(response_body);
444 return true;
445}
