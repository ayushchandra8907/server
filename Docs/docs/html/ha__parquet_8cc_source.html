257static std::string item_to_sql(const Item *item)
258{
259 if (!item) return "";
260 const Item_func *func = dynamic_cast<const Item_func *>(item);
261 if (!func || func->argument_count() != 2) return "";
262
263
264 const Item *left = func->arguments()[0];
265 const Item *right = func->arguments()[1];
266
267
268 if (left->type() != Item::FIELD_ITEM) return "";
269
270
271 const Item_field *field = static_cast<const Item_field *>(left);
272 std::string col = field->field_name.str;
273 std::string val;
274
275
276 if (right->type() == Item::CONST_ITEM) {
277 String tmp;
278 String *s = const_cast<Item*>(right)->val_str(&tmp);
279 if (!s) return "";
280 std::string strval(s->ptr(), s->length());
281 bool is_number = !strval.empty() &&
282 (isdigit((unsigned char)strval[0]) || strval[0] == '-');
283 val = is_number ? strval : quote_string_literal(strval);
284 } else {
285 return "";
286 }
287
288
289 std::string op = func->func_name();
290 if (op == "=" || op == "eq") return col + " = " + val;
291 if (op == ">" || op == "gt") return col + " > " + val;
292 if (op == "<" || op == "lt") return col + " < " + val;
293 return "";
294}

359{
360 std::string query = "CREATE TABLE IF NOT EXISTS " + table_name + " (";

369 }
370 query += ")";

518static bool parquet_stage_statement_buffer_to_local(
519 parquet_table_trx_data *table_trx, const std::string &canonical_parquet_path,
520 std::string *error)
521{
522 if (table_trx == nullptr || table_trx->statement_row_count == 0 ||
523 table_trx->statement_buffer_name.empty()) {
524 return true;
525 }
526
527 const auto stage_id = parquet_next_local_stage_id();
528 const std::string local_stage_path =
529 parquet::BuildLocalStagePath(canonical_parquet_path, stage_id);
530
531 duckdb::Connection *con = nullptr;
532 try {
533 std::lock_guard<std::mutex> lock(g_duckdb_mutex);
534 con = new duckdb::Connection(*g_duckdb);
535 parquet_log_info("DuckDB query [txn/stage-local/install] INSTALL parquet;");
536 con->Query("INSTALL parquet;");
537 parquet_log_info("DuckDB query [txn/stage-local/load] LOAD parquet;");
538 con->Query("LOAD parquet;");
539 std::remove(local_stage_path.c_str());
540 const std::string copy_query = build_copy_to_parquet_query(
541 table_trx->statement_buffer_name, local_stage_path);
542 parquet_log_info("DuckDB query [txn/stage-local/copy] " +
543 parquet_log_preview(copy_query));
544 auto copy_result = con->Query(copy_query);
545 if (!copy_result || copy_result->HasError()) {
546 if (error != nullptr) {
547 *error = copy_result ? copy_result->GetError()
548 : "DuckDB returned a null result while staging";
549 }
550 delete con;
551 return false;
552 }
553 delete con;
554 } catch (const std::exception &e) {
555 delete con;
556 if (error != nullptr) {
557 *error = e.what();
558 }
559 return false;
560 }
561
562 table_trx->staged_files.push_back(
563 {local_stage_path, table_trx->statement_row_count});
564 parquet_log_info("Parquet staged local file table='" + table_trx->table_name +
565 "' path='" + local_stage_path + "' rows=" +
566 std::to_string(table_trx->statement_row_count));
567 parquet_reset_statement_buffer(table_trx);
568 return true;
569}

637 parquet_log_preview(copy_query));

655 metadata_result->RowCount() > 0 &&
656 !metadata_result->GetValue(0, 0).IsNull()) {
657 compressed_size = metadata_result->GetValue(0, 0).GetValue<int64_t>();
658 }
659 delete con;
660 } catch (const std::exception &e) {
661 delete con;
662 if (error != nullptr) {
663 *error = e.what();

666 }

670 table_trx->table_name + "' rows=" +
671 std::to_string(total_rows) + " bytes=" +
672 std::to_string(compressed_size));
673
674 if (s3_path != nullptr) {
675 *s3_path = target_s3_path;
676 }
677 if (record_count != nullptr) {
678 *record_count = total_rows;
679 }
680 if (file_size != nullptr) {
681 *file_size = compressed_size;
682 }
683 return true;

1100{
1101 (void) buf;
1102 DBUG_ENTER("ha_parquet::rnd_next");
1103
1104
1105 if (!scan_result || current_row >= scan_result->RowCount())
1106 DBUG_RETURN(HA_ERR_END_OF_FILE);

1108
1109 for (uint i = 0; i < table->s->fields; i++) {
1110 Field *f = table->field[i];
1111 auto val = scan_result->GetValue(i, current_row);

1114 f->set_null();

1116 }

1118 f->set_notnull();
1119
1120
1121 enum_field_types t = f->type();
1122 if (t == MYSQL_TYPE_TINY || t == MYSQL_TYPE_SHORT ||

1124 f->store((longlong)val.GetValue<int32_t>(), false);
1125 else if (t == MYSQL_TYPE_LONGLONG)
1126 f->store((longlong)val.GetValue<int64_t>(), false);
1127 else if (t == MYSQL_TYPE_FLOAT || t == MYSQL_TYPE_DOUBLE)
1128 f->store(val.GetValue<double>());
1129 else {
1130 std::string s = val.ToString();
1131 f->store(s.c_str(), s.length(), f->charset());
1132 }
1133 }
1134
1135
1136 current_row++;

1243 if (!parquet_is_real_commit(thd, all)) {
1244 if (has_pending_work) {
1245 parquet_log_info(
1246 "Parquet statement commit complete; remote flush deferred until "
1247 "transaction commit tables=" +
1248 std::to_string(pending_table_count));
1249 }
1250 return 0;
1251 }
1252
1253 if (!has_pending_work) {
1254 parquet_log_info("Parquet transaction commit has no staged write work");
1255 delete trx;
1256 thd_set_ha_data(thd, parquet_hton, nullptr);
1257 return 0;
1258 }
1259
1260 if (pending_table_count > 1) {
1261 raise_unknown_error(
1262 "Parquet currently supports committing writes to only one table per "
1263 "transaction");
1264 return 1;
1265 }
1266
1267 parquet_log_info("Parquet transaction commit begin tables=" +
1268 std::to_string(pending_table_count));
1269
1270 static long long snapshot_counter = 0;
1271 for (auto &entry : trx->tables) {
1272 parquet_table_trx_data &table_trx = entry.second;
1273 if (table_trx.staged_files.empty()) {
1274 continue;
1275 }
1276
1277 parquet::TableMetadata metadata;
1278 if (!resolve_runtime_metadata_or_error(table_trx.table_path.c_str(),
1279 &metadata) ||
1280 !validate_catalog_or_error(metadata) ||
1281 !validate_object_store_or_error(metadata)) {
1282 return 1;
1283 }
1284
1285 std::string s3_path;
1286 int64_t record_count = 0;
1287 int64_t file_size = 0;
1288 std::string flush_error;
1289 if (!parquet_flush_local_stages_to_s3(&table_trx, metadata, &s3_path,
1290 &record_count, &file_size,
1291 &flush_error)) {
1292 raise_unknown_error("Failed to flush Parquet rows to S3 during commit: " +
1293 flush_error);
1294 return 1;
1295 }
1296
1297 parquet_log_info("Parquet commit uploaded staged files table='" +
1298 table_trx.table_name + "' staged_file_count=" +
1299 std::to_string(table_trx.staged_files.size()) +
1300 " target='" + s3_path + "'");
1301 long long snapshot_id =
1302 static_cast<long long>(time(NULL)) * 1000 +
1303 (snapshot_counter++ % 1000);
1304 json snapshot_summary = {{"operation", "append"},
1305 {"added-data-files", "1"},
1306 {"added-records",
1307 std::to_string(record_count)},
1308 {"added-files-size",
1309 std::to_string(file_size)}};
1310 json snapshot = {{"snapshot-id", snapshot_id},
1311 {"sequence-number", 1},
1312 {"timestamp-ms", snapshot_id * 1000},
1313 {"summary", snapshot_summary},
1314 {"schema-id", 0},
1315 {"manifest-list", s3_path}};
1316 json updates = json::array();
1317 updates.push_back({{"action", "add-snapshot"}, {"snapshot", snapshot}});
1318 updates.push_back({{"action", "set-snapshot-ref"},
1319 {"ref-name", "main"},
1320 {"type", "branch"},
1321 {"snapshot-id", snapshot_id}});
1322 json identifier = {{"namespace",
1323 namespace_json_array(
1324 metadata.catalog_table_ident.namespace_ident)},
1325 {"name", metadata.catalog_table_ident.table_name}};
1326 json table_change = {{"identifier", identifier},
1327 {"requirements", json::array()},
1328 {"updates", updates}};
1329 const std::string json_body =
1330 json({{"table-changes", json::array({table_change})}}).dump();
1331
1332 CURL *curl = curl_easy_init();
1333 if (!curl) {
1334 return 1;
1335 }
1336
1337 struct curl_slist *headers = NULL;
1338 headers = curl_slist_append(headers, "Content-Type: application/json");
1339 curl_easy_setopt(curl, CURLOPT_POST, 1L);

1341 lakekeeper_transaction_commit_url(metadata);
1342 curl_easy_setopt(curl, CURLOPT_URL, lakekeeper_url.c_str());
1343 curl_easy_setopt(curl, CURLOPT_POSTFIELDS, json_body.c_str());
1344 apply_catalog_request_options(metadata.catalog_config, curl, &headers);
1345 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers);
1346 parquet_log_info("LakeKeeper commit begin table='" + table_trx.table_name +
1347 "' snapshot_id=" + std::to_string(snapshot_id) +
1348 " manifest='" + s3_path + "' url='" + lakekeeper_url +
1349 "' payload=" + parquet_log_preview(json_body));
1350
1351 CURLcode res = curl_easy_perform(curl);
1352 long http_code = 0;
1353 curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, &http_code);
1354 curl_slist_free_all(headers);
1355 curl_easy_cleanup(curl);
1356
1357 if (res != CURLE_OK) {
1358 std::cerr << "LakeKeeper commit error: " << curl_easy_strerror(res)
1359 << std::endl;
1360 return 1;
1361 }

1363 std::cerr << "LakeKeeper commit HTTP error: " << http_code
1364 << std::endl;
1365 return 1;
1366 }
1367
1368 parquet_log_info("LakeKeeper commit complete table='" +
1369 table_trx.table_name + "' snapshot_id=" +
1370 std::to_string(snapshot_id) + " http_status=" +
1371 std::to_string(http_code));
1372 parquet_remove_local_stage_files(&table_trx);
1373 table_trx.uploaded_s3_file_paths.clear();
1374 }
1375
1376 parquet_log_info("Parquet transaction commit complete");
1377 delete trx;
1378 thd_set_ha_data(thd, parquet_hton, nullptr);
1379 return 0;
1380}

1383static int ha_parquet_rollback(THD *thd, bool all)
1384{
1385 parquet_trx_data *trx =
1386 static_cast<parquet_trx_data *>(thd_get_ha_data(thd, parquet_hton));
1387 if (!trx) return 0;
1388
1389 const bool real_rollback = parquet_is_real_rollback(thd, all);
1390 parquet_log_info(std::string("Parquet rollback begin scope='") +
1391 (real_rollback ? "transaction" : "statement") + "'");
1392 int error = 0;
1393 for (auto &entry : trx->tables) {
1394 parquet_table_trx_data &table_trx = entry.second;
1395 parquet_reset_statement_buffer(&table_trx);
1396
1397 if (!real_rollback) {
1398 continue;
1399 }
1400
1401 parquet_remove_local_stage_files(&table_trx);
1402 if (table_trx.uploaded_s3_file_paths.empty()) {
1403 continue;
1404 }
1405
1406 parquet::TableMetadata metadata;
1407 if (!resolve_runtime_metadata_or_error(table_trx.table_path.c_str(),
1408 &metadata) ||
1409 !validate_object_store_or_error(metadata)) {
1410 error = 1;
1411 continue;
1412 }
1413
1414 parquet::ParquetObjectStoreClient object_store(
1415 metadata.object_store_config);
1416 for (const std::string &s3_path : table_trx.uploaded_s3_file_paths) {
1417 parquet::ObjectLocation location;
1418 if (!parquet::ParseS3Uri(s3_path, &location)) {
1419 std::cerr << "rollback: failed to parse staged object path "
1420 << s3_path << std::endl;
1421 error = 1;
1422 continue;
1423 }
1424 location = parquet::ResolveAbsoluteObjectLocation(
1425 metadata.object_store_config, location.bucket, location.key);
1426
1427 parquet_log_info("Parquet rollback deleting uploaded object table='" +
1428 table_trx.table_name + "' path='" + s3_path + "'");
1429 auto status = object_store.DeleteObject(location);
1430 if (!status.ok()) {
1431 std::cerr << "rollback: failed to delete " << s3_path << ": "
1432 << status.message << " (HTTP " << status.http_status
1433 << ")" << std::endl;
1434 error = 1;
1435 }
1436 }
1437 }
1438
1439 if (real_rollback) {
1440 delete trx;
1441 thd_set_ha_data(thd, parquet_hton, nullptr);
1442 }
1443 parquet_log_info("Parquet rollback complete");
1444 return error;
1445}
