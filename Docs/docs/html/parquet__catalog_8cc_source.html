50std::string TrimAsciiWhitespace(const std::string &input)
51{
52 size_t begin = 0;
53 while (begin < input.size() &&
54 std::isspace(static_cast<unsigned char>(input[begin]))) {
55 begin++;
56 }
57
58 size_t end = input.size();
59 while (end > begin &&
60 std::isspace(static_cast<unsigned char>(input[end - 1]))) {
61 end--;
62 }
63
64 return input.substr(begin, end - begin);
65}

76size_t HeaderCallback(char *buffer, size_t size, size_t nitems, void *userdata)
77{
78 auto *headers = static_cast<std::map<std::string, std::string> *>(userdata);
79 const auto bytes = size * nitems;
80 std::string line(buffer, bytes);
81 const auto colon = line.find(':');
82
83 if (colon != std::string::npos) {
84 auto key = ToLowerAscii(TrimAsciiWhitespace(line.substr(0, colon)));
85 auto value = TrimAsciiWhitespace(line.substr(colon + 1));
86 headers->insert_or_assign(key, value);
87 }
88
89 return bytes;
90}

97std::string PercentEncode(const std::string &value)
98{
99 static const char kHex[] = "0123456789ABCDEF";
100
101 std::string encoded;
102 encoded.reserve(value.size() * 3);
103
104 for (unsigned char ch : value) {
105 if (IsUnreservedUrlByte(ch)) {
106 encoded.push_back(static_cast<char>(ch));
107 continue;
108 }
109
110 encoded.push_back('%');
111 encoded.push_back(kHex[(ch >> 4) & 0x0F]);
112 encoded.push_back(kHex[ch & 0x0F]);
113 }
114
115 return encoded;
116}

153CatalogStatus HttpStatusToCatalogStatus(long http_status,
154 const std::string &message)
155{
156 if (http_status >= 200 && http_status < 300) {
157 return MakeStatus(CatalogStatusCode::kOk, http_status, false, false,
158 message);
159 }
160
161 if (http_status == 400) {
162 return MakeStatus(CatalogStatusCode::kInvalidArgument, http_status, false,
163 false, message);
164 }
165
166 if (http_status == 401) {
167 return MakeStatus(CatalogStatusCode::kUnauthorized, http_status, false,
168 false, message);
169 }
170
171 if (http_status == 403) {
172 return MakeStatus(CatalogStatusCode::kForbidden, http_status, false, false,
173 message);
174 }
175
176 if (http_status == 404) {
177 return MakeStatus(CatalogStatusCode::kNotFound, http_status, false, false,
178 message);
179 }
180
181 if (http_status == 409) {
182 return MakeStatus(CatalogStatusCode::kConflict, http_status, true, false,
183 message);
184 }
185
186 return MakeStatus(CatalogStatusCode::kServerError, http_status,
187 http_status >= 500, false, message);
188}

241HttpResponse ExecuteRequest(const CatalogClientConfig &config,
242 const std::string &method,
243 const std::string &url,
244 const std::string &body,
245 const std::vector<std::string> &headers)
246{
247 EnsureCurlInitialized();
248 parquet_log_info("LakeKeeper request method='" + method + "' url='" + url +
249 "' body=" + parquet_log_preview(body));
250
251 HttpResponse response;
252 CURL *curl = curl_easy_init();
253 if (curl == nullptr) {
254 response.curl_code = CURLE_FAILED_INIT;
255 response.curl_error = "curl_easy_init failed";
256 return response;
257 }
258
259 char error_buffer[CURL_ERROR_SIZE] = {0};
260 struct curl_slist *curl_headers = nullptr;
261
262 curl_easy_setopt(curl, CURLOPT_ERRORBUFFER, error_buffer);
263 curl_easy_setopt(curl, CURLOPT_URL, url.c_str());
264 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, method.c_str());
265
266 ApplyCommonCurlOptions(curl, config, &response);
267
268 if (!body.empty()) {
269 curl_easy_setopt(curl, CURLOPT_POSTFIELDS, body.c_str());
270 curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE,
271 static_cast<long>(body.size()));
272 }
273
274 for (const auto &header : headers) {
275 curl_headers = curl_slist_append(curl_headers, header.c_str());
276 }
277
278 if (curl_headers != nullptr) {
279 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, curl_headers);
280 }
281
282 response.curl_code = curl_easy_perform(curl);
283 if (response.curl_code != CURLE_OK) {
284 response.curl_error = error_buffer[0] ? error_buffer
285 : curl_easy_strerror(response.curl_code);
286 }
287
288 curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, &response.status_code);
289
290 if (response.curl_code == CURLE_OK) {
291 parquet_log_info("LakeKeeper response method='" + method + "' url='" + url +
292 "' http_status=" + std::to_string(response.status_code) +
293 " body=" + parquet_log_preview(response.body));
294 } else {
295 parquet_log_warning("LakeKeeper request failed method='" + method +
296 "' url='" + url + "' error='" +
297 response.curl_error + "'");
298 }
299
300 curl_slist_free_all(curl_headers);
301 curl_easy_cleanup(curl);
302 return response;
303}

305std::string JoinPath(const std::string &lhs, const std::string &rhs)
306{
307 if (lhs.empty()) {
308 return rhs;
309 }
310
311 if (rhs.empty()) {
312 return lhs;
313 }
314
315 if (lhs.back() == '/' && rhs.front() == '/') {
316 return lhs + rhs.substr(1);
317 }
318
319 if (lhs.back() != '/' && rhs.front() != '/') {
320 return lhs + "/" + rhs;
321 }
322
323 return lhs + rhs;
324}

678CatalogCapabilitySet ResolveCatalogCapabilities(
679 const std::vector<std::string> &advertised_endpoints)
680{
681 CatalogCapabilitySet capabilities;
682 capabilities.advertised_endpoints = advertised_endpoints;
683
684 if (advertised_endpoints.empty()) {
685 capabilities.supports_create_table = true;
686 capabilities.supports_commit_table = true;
687 capabilities.supports_commit_transaction = true;
688 capabilities.supports_register_table = true;
689 capabilities.supports_scan_planning = false;
690 return capabilities;
691 }
692
693 capabilities.supports_create_table = false;
694 capabilities.supports_commit_table = false;
695 capabilities.supports_commit_transaction = false;
696 capabilities.supports_register_table = false;
697 capabilities.supports_scan_planning = false;
698
699 for (const auto &endpoint : advertised_endpoints) {
700 if (EndpointMatches(endpoint, "POST",
701 "/namespaces/{namespace}/tables")) {
702 capabilities.supports_create_table = true;
703 }
704
705 if (EndpointMatches(endpoint, "POST",
706 "/namespaces/{namespace}/tables/{table}")) {
707 capabilities.supports_commit_table = true;
708 }
709
710 if (EndpointMatches(endpoint, "POST", "/transactions/commit")) {
711 capabilities.supports_commit_transaction = true;
712 }
713
714 if (EndpointMatches(endpoint, "POST",
715 "/namespaces/{namespace}/register")) {
716 capabilities.supports_register_table = true;
717 }
718
719 if (EndpointMatches(endpoint, "POST",
720 "/namespaces/{namespace}/tables/{table}/plan")) {
721 capabilities.supports_scan_planning = true;
722 }
723 }
724

726}

752CatalogStatus ParquetCatalogClient::BootstrapConfig()
753{
754 std::string url = JoinPath(config_.base_uri, "/v1/config");
755 if (!config_.warehouse.empty()) {
756 url += "?warehouse=" + PercentEncode(config_.warehouse);
757 }
758
759 auto headers = BuildJsonHeaders(config_, "", "", "", "");
760 auto response = ExecuteRequest(config_, "GET", url, "", headers);
761 auto status = ParseBasicSuccessOrError(response);
762 if (!status.ok()) {
763 return status;
764 }
765
766 try {
767 auto payload = json::parse(response.body.empty() ? "{}" : response.body);
768 effective_config_.clear();
769
770 if (payload.contains("defaults")) {
771 effective_config_ = JsonObjectToStringMap(payload["defaults"]);
772 }
773
774 for (const auto &entry : config_.client_properties) {
775 effective_config_[entry.first] = entry.second;
776 }
777
778 if (payload.contains("overrides")) {
779 auto overrides = JsonObjectToStringMap(payload["overrides"]);
