58static duckdb::LogicalType field_to_logical_type(const Field *field)
59{
60 const bool is_unsigned = (field->flags & UNSIGNED_FLAG) != 0;
61
62 switch (field->real_type()) {
63 case MYSQL_TYPE_TINY:
64 return is_unsigned ? duckdb::LogicalType::UTINYINT
65 : duckdb::LogicalType::TINYINT;
66 case MYSQL_TYPE_SHORT:
67 return is_unsigned ? duckdb::LogicalType::USMALLINT
68 : duckdb::LogicalType::SMALLINT;
69 case MYSQL_TYPE_INT24:
70 case MYSQL_TYPE_LONG:
71 return is_unsigned ? duckdb::LogicalType::UINTEGER
72 : duckdb::LogicalType::INTEGER;
73 case MYSQL_TYPE_LONGLONG:
74 return is_unsigned ? duckdb::LogicalType::UBIGINT
75 : duckdb::LogicalType::BIGINT;
76 case MYSQL_TYPE_FLOAT:
77 return duckdb::LogicalType::FLOAT;
78 case MYSQL_TYPE_DOUBLE:
79 return duckdb::LogicalType::DOUBLE;
80 case MYSQL_TYPE_DECIMAL:
81 case MYSQL_TYPE_NEWDECIMAL: {
82 auto *df = static_cast<const Field_new_decimal *>(field);
83 uint prec = df->precision > 38 ? 38 : df->precision;
84 return duckdb::LogicalType::DECIMAL(prec, df->dec);
85 }
86 case MYSQL_TYPE_DATE:
87 case MYSQL_TYPE_NEWDATE:
88 return duckdb::LogicalType::DATE;
89 case MYSQL_TYPE_TIME:
90 case MYSQL_TYPE_TIME2:
91 return duckdb::LogicalType::TIME;
92 case MYSQL_TYPE_DATETIME:
93 case MYSQL_TYPE_DATETIME2:
94 return duckdb::LogicalType::TIMESTAMP;
95 case MYSQL_TYPE_TIMESTAMP:
96 case MYSQL_TYPE_TIMESTAMP2:
97 return duckdb::LogicalType::TIMESTAMP_TZ;
98 case MYSQL_TYPE_YEAR:
99 return duckdb::LogicalType::INTEGER;
100 case MYSQL_TYPE_BIT:
101 case MYSQL_TYPE_GEOMETRY:
102 return duckdb::LogicalType::BLOB;
103 case MYSQL_TYPE_BLOB:
104 case MYSQL_TYPE_TINY_BLOB:
105 case MYSQL_TYPE_MEDIUM_BLOB:
106 case MYSQL_TYPE_LONG_BLOB:
107 return field->has_charset() ? duckdb::LogicalType::VARCHAR
108 : duckdb::LogicalType::BLOB;
109 case MYSQL_TYPE_STRING:
110 case MYSQL_TYPE_VARCHAR:
111 case MYSQL_TYPE_VAR_STRING:
112 case MYSQL_TYPE_SET:
113 case MYSQL_TYPE_ENUM:
114 return duckdb::LogicalType::VARCHAR;
115 default:
116 return duckdb::LogicalType::VARCHAR;
117 }
118}

120static duckdb::Value field_to_duckdb_value(Field *field)
121{
122 if (field->is_null())
123 return duckdb::Value();
124
125 switch (field->real_type()) {
126 case MYSQL_TYPE_TINY:
127 case MYSQL_TYPE_SHORT:
128 case MYSQL_TYPE_INT24:
129 case MYSQL_TYPE_LONG:
130 case MYSQL_TYPE_LONGLONG:
131 case MYSQL_TYPE_YEAR:
132 if (field->is_unsigned())
133 return duckdb::Value::UBIGINT(field->val_uint());
134 return duckdb::Value::BIGINT(field->val_int());
135 case MYSQL_TYPE_FLOAT:
136 return duckdb::Value::FLOAT(static_cast<float>(field->val_real()));
137 case MYSQL_TYPE_DOUBLE:
138 return duckdb::Value::DOUBLE(field->val_real());
139 case MYSQL_TYPE_DECIMAL:
140 case MYSQL_TYPE_NEWDECIMAL:
141 case MYSQL_TYPE_DATE:
142 case MYSQL_TYPE_NEWDATE:
143 case MYSQL_TYPE_TIME:
144 case MYSQL_TYPE_TIME2:
145 case MYSQL_TYPE_DATETIME:
146 case MYSQL_TYPE_DATETIME2:
147 case MYSQL_TYPE_TIMESTAMP:
148 case MYSQL_TYPE_TIMESTAMP2:
149 default: {
150 String buf;
151 field->val_str(&buf);
152 return duckdb::Value(std::string(buf.ptr(), buf.length()));
153 }
154 case MYSQL_TYPE_BIT:
155 case MYSQL_TYPE_GEOMETRY: {
156 String buf;
157 field->val_str(&buf);
158 return duckdb::Value::BLOB(std::string(buf.ptr(), buf.length()));
159 }
160 case MYSQL_TYPE_BLOB:
161 case MYSQL_TYPE_TINY_BLOB:
162 case MYSQL_TYPE_MEDIUM_BLOB:
163 case MYSQL_TYPE_LONG_BLOB: {
164 String buf;
165 field->val_str(&buf);
166 if (field->has_charset())
167 return duckdb::Value(std::string(buf.ptr(), buf.length()));
168 return duckdb::Value::BLOB(std::string(buf.ptr(), buf.length()));
169 }
170 }

208static bool cast_value_for_filter(const duckdb::Value &source,
209 const duckdb::LogicalType &target_type,
210 duckdb::Value *cast_value)
211{
212 if (!cast_value)
213 return false;
214
215 if (source.IsNull()) {
216 *cast_value = duckdb::Value(target_type);
217 return true;
218 }
219
220 if (source.type() == target_type) {
221 *cast_value = source;
222 return true;
223 }
224
225 std::string error_message;
226 return source.DefaultTryCastAs(target_type, *cast_value, &error_message);
227}

245static bool evaluate_in_filter(const duckdb::InFilter &filter,
246 const duckdb::Value &row_value)
247{
248 if (row_value.IsNull() || filter.values.empty())
249 return false;
250
251 duckdb::Value comparable_value;
252 if (!cast_value_for_filter(row_value, filter.values[0].type(), &comparable_value))
253 return false;
254
255 for (const auto &candidate : filter.values) {
256 if (duckdb::Value::NotDistinctFrom(comparable_value, candidate))
257 return true;
258 }
259 return false;
260}

262static bool evaluate_filter_on_value(const duckdb::TableFilter &filter,
263 const duckdb::Value &row_value)
264{
265 switch (filter.filter_type) {
266 case duckdb::TableFilterType::CONSTANT_COMPARISON: {
267 const auto &constant_filter = filter.Cast<duckdb::ConstantFilter>();
268 if (row_value.IsNull())
269 return false;
270
271 duckdb::Value comparable_value;
272 if (!cast_value_for_filter(row_value, constant_filter.constant.type(),
273 &comparable_value)) {
274 return false;
275 }
276 return constant_filter.Compare(comparable_value);
277 }
278 case duckdb::TableFilterType::IS_NULL:
279 return row_value.IsNull();
280 case duckdb::TableFilterType::IS_NOT_NULL:
281 return !row_value.IsNull();
282 case duckdb::TableFilterType::CONJUNCTION_AND: {
283 const auto &and_filter = filter.Cast<duckdb::ConjunctionAndFilter>();
284 for (const auto &child : and_filter.child_filters) {
285 if (child && !evaluate_filter_on_value(*child, row_value))
286 return false;
287 }
288 return true;
289 }
290 case duckdb::TableFilterType::CONJUNCTION_OR: {
291 const auto &or_filter = filter.Cast<duckdb::ConjunctionOrFilter>();
292 for (const auto &child : or_filter.child_filters) {
293 if (child && evaluate_filter_on_value(*child, row_value))
294 return true;
295 }
296 return false;
297 }
298 case duckdb::TableFilterType::OPTIONAL_FILTER: {
299 const auto &optional_filter = filter.Cast<duckdb::OptionalFilter>();
300 return !optional_filter.child_filter ||
301 evaluate_filter_on_value(*optional_filter.child_filter, row_value);
302 }
303 case duckdb::TableFilterType::IN_FILTER:
304 return evaluate_in_filter(filter.Cast<duckdb::InFilter>(), row_value);
305 case duckdb::TableFilterType::DYNAMIC_FILTER:
306 return evaluate_dynamic_filter(filter.Cast<duckdb::DynamicFilter>(), row_value);
307 default:
308 throw duckdb::NotImplementedException(
309 "Parquet cross-engine filter pushdown does not support filter type %d",
310 static_cast<int>(filter.filter_type));
311 }
312}

314static bool row_passes_filters(TABLE *tbl, const duckdb::TableFilterSet &filters)
315{
316 for (const auto &entry : filters.filters) {
317 if (entry.first >= tbl->s->fields)
318 return false;
319
320 Field *field = tbl->field[entry.first];
321 duckdb::Value row_value = field_to_duckdb_value(field);
322 if (!evaluate_filter_on_value(*entry.second, row_value))
323 return false;
324 }
325 return true;
326}

328static std::string describe_filter_set(TABLE *tbl,
329 const duckdb::TableFilterSet &filters)
330{
331 std::string description;
332 for (const auto &entry : filters.filters) {
333 if (!description.empty())
334 description += ", ";
335
336 const char *field_name =
337 (tbl && entry.first < tbl->s->fields && tbl->field[entry.first])
338 ? tbl->field[entry.first]->field_name.str
339 : "<unknown>";
340 description += entry.second->ToString(field_name);
341 }
342 return description;
343}

388static void mdb_scan_function(duckdb::ClientContext &context,
389 duckdb::TableFunctionInput &input,
390 duckdb::DataChunk &output)
391{
392 (void) context;
393 auto &state = input.global_state->Cast<MdbScanGlobalState>();
394
395 if (state.finished) {
396 if (!state.completion_logged) {
397 sql_print_information("Parquet: _mdb_scan complete table='%s' scanned=%llu emitted=%llu filtered=%llu",
398 state.table_key.c_str(),
399 static_cast<unsigned long long>(state.rows_scanned),
400 static_cast<unsigned long long>(state.rows_emitted),
401 static_cast<unsigned long long>(state.rows_filtered));
402 state.completion_logged = true;
403 }
404 output.SetCardinality(0);
405 return;
406 }
407
408 TABLE *tbl = state.table;
409 if (!tbl) {
410 output.SetCardinality(0);
411 state.finished = true;
412 if (!state.completion_logged) {
413 sql_print_information("Parquet: _mdb_scan complete table='%s' scanned=%llu emitted=%llu filtered=%llu",
414 state.table_key.c_str(),
415 static_cast<unsigned long long>(state.rows_scanned),
416 static_cast<unsigned long long>(state.rows_emitted),
417 static_cast<unsigned long long>(state.rows_filtered));
418 state.completion_logged = true;
419 }
420 return;
421 }
422
423 THD *prev_thd = _current_thd();
424 if (tbl->in_use && tbl->in_use != prev_thd)
425 set_current_thd(tbl->in_use);
426
427 if (!state.scan_started) {
428 sql_print_information("Parquet: _mdb_scan start table='%s' projected_columns=%zu",
429 state.table_key.c_str(), state.column_ids.size());
430 if (state.filters && !state.filters->filters.empty()) {
431 const std::string filter_desc = describe_filter_set(tbl, *state.filters);
432 sql_print_information("Parquet: _mdb_scan pushed_filters table='%s' filters=[%s]",
433 state.table_key.c_str(), filter_desc.c_str());
434 }
435 bitmap_clear_all(tbl->read_set);
436 for (auto col_idx : state.column_ids)
437 bitmap_set_bit(tbl->read_set, static_cast<uint>(col_idx));
438
439 if (tbl->file->ha_rnd_init(true)) {
440 sql_print_warning("Parquet: _mdb_scan failed to initialize table='%s'",
441 state.table_key.c_str());
442 state.finished = true;
443 output.SetCardinality(0);
444 if (!state.completion_logged) {
445 sql_print_information("Parquet: _mdb_scan complete table='%s' scanned=%llu emitted=%llu filtered=%llu",
446 state.table_key.c_str(),
447 static_cast<unsigned long long>(state.rows_scanned),
448 static_cast<unsigned long long>(state.rows_emitted),
449 static_cast<unsigned long long>(state.rows_filtered));
450 state.completion_logged = true;
451 }
452 if (_current_thd() != prev_thd)
453 set_current_thd(prev_thd);
454 return;
455 }
456 state.scan_started = true;
457 }
458
459 duckdb::idx_t count = 0;
460 duckdb::idx_t ncols = state.column_ids.size();
461
462 while (count < STANDARD_VECTOR_SIZE) {
463 int err = tbl->file->ha_rnd_next(tbl->record[0]);
464 if (err) {
465 tbl->file->ha_rnd_end();
466 state.finished = true;
467 break;
468 }
469
470 state.rows_scanned++;
471
472 if (state.filters && !state.filters->filters.empty() &&
473 !row_passes_filters(tbl, *state.filters)) {
474 state.rows_filtered++;
475 continue;
476 }
477
478 for (duckdb::idx_t i = 0; i < ncols; ++i) {
479 Field *field = tbl->field[state.column_ids[i]];
480 duckdb::Value val = field_to_duckdb_value(field);
481 output.data[i].SetValue(count, val);
482 }
483 count++;
484 }
485
486 state.rows_emitted += count;
487 output.SetCardinality(count);
488
489 if (state.finished && !state.completion_logged) {
490 sql_print_information("Parquet: _mdb_scan complete table='%s' scanned=%llu emitted=%llu filtered=%llu",
491 state.table_key.c_str(),
492 static_cast<unsigned long long>(state.rows_scanned),
493 static_cast<unsigned long long>(state.rows_emitted),
494 static_cast<unsigned long long>(state.rows_filtered));
495 state.completion_logged = true;
496 }
497
498 if (_current_thd() != prev_thd)
499 set_current_thd(prev_thd);
500}
