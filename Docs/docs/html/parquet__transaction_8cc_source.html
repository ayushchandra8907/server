80bool ValidateTxnState(const ParquetTxnState &txn_state, std::string *error)
81{
82 if (txn_state.has_error) {
83 if (error != nullptr) {
84 *error = "transaction state is marked as failed";
85 }
86 return false;
87 }
88
89 if (!txn_state.staged_files.empty() && !txn_state.registered_with_server) {
90 if (error != nullptr) {
91 *error = "transaction state has staged files but was never registered";
92 }
93 return false;
94 }
95
96 for (const auto &staged_file : txn_state.staged_files) {
97 if (!IsStagedFileComplete(staged_file)) {
98 if (error != nullptr) {
99 *error = "transaction state contains an incomplete staged file entry";
100 }
101 return false;
102 }
103 }
104
105 return true;
106}
