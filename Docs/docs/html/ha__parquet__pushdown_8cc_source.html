22std::string quote_identifier(const std::string &identifier)
23{
24 std::string quoted = "\"";
25 for (char ch : identifier) {
26 if (ch == '"')
27 quoted += "\"\"";
28 else
29 quoted += ch;
30 }
31 quoted += "\"";
32 return quoted;

75 return "TIMESTAMP";

85std::string build_empty_view_query(TABLE_LIST *tbl)
86{
87 std::string query = "CREATE OR REPLACE TEMP VIEW " +
88 quote_identifier(tbl->table_name.str) + " AS SELECT ";
89 bool first = true;
90
91 for (Field **field = tbl->table->field; *field; ++field) {
92 Field *f = *field;
93 const std::string duck_type = mariadb_type_to_duckdb(f);
94 if (duck_type.empty())
95 return "";
96
97 if (!first)
98 query += ", ";
99 first = false;
100 query += "CAST(NULL AS " + duck_type + ") AS " +
101 quote_identifier(f->field_name.str);
102 }
103
104 query += " WHERE FALSE";
105 return query;
106}

108std::string build_read_view_query(TABLE_LIST *tbl,
109 const std::vector<std::string> &s3_files)
110{
111 std::string parquet_file_list = "[";
112 for (size_t i = 0; i < s3_files.size(); ++i) {
113 if (i != 0)
114 parquet_file_list += ", ";
115 parquet_file_list += quote_string_literal(s3_files[i]);
116 }
117 parquet_file_list += "]";
118
119 return "CREATE OR REPLACE TEMP VIEW " + quote_identifier(tbl->table_name.str) +
120 " AS SELECT * FROM read_parquet(" + parquet_file_list + ")";

128void store_duckdb_field_in_mysql_format(Field *field, duckdb::Value &value,
129 THD *thd)
130{
131 (void) thd;
132 if (value.IsNull()) {
133 field->set_default();
134 if (field->real_maybe_null())
135 field->set_null();
136 return;
137 }
138
139 field->set_notnull();
140 switch (field->type()) {
141 case MYSQL_TYPE_TINY_BLOB:
142 case MYSQL_TYPE_MEDIUM_BLOB:
143 case MYSQL_TYPE_LONG_BLOB:
144 case MYSQL_TYPE_BLOB:
145 case MYSQL_TYPE_GEOMETRY:
146 case MYSQL_TYPE_BIT: {
147 auto str = value.GetValueUnsafe<duckdb::string>();
148 field->store(str.c_str(), str.size(), &my_charset_bin);
149 break;
150 }
151 case MYSQL_TYPE_VARCHAR:
152 case MYSQL_TYPE_STRING:
153 case MYSQL_TYPE_VAR_STRING: {
154 auto str = value.GetValue<duckdb::string>();
155 field->store(str.c_str(), str.size(),
156 field->has_charset() ? field->charset() : &my_charset_bin);
157 break;
158 }
159 case MYSQL_TYPE_NULL:
160 case MYSQL_TYPE_DECIMAL:
161 case MYSQL_TYPE_ENUM:
162 case MYSQL_TYPE_SET:
163 case MYSQL_TYPE_NEWDECIMAL: {
164 auto str = value.GetValue<duckdb::string>();
165 field->store(str.c_str(), str.size(), system_charset_info);
166 break;
167 }
168 case MYSQL_TYPE_TINY:
169 case MYSQL_TYPE_YEAR:
170 case MYSQL_TYPE_SHORT:
171 case MYSQL_TYPE_INT24:
172 case MYSQL_TYPE_LONG: {
173 int64_t v = value.GetValue<int64_t>();
174 field->store(v, field->is_unsigned());
175 break;
176 }
177 case MYSQL_TYPE_LONGLONG: {
178 if (field->is_unsigned())
179 field->store(value.GetValue<uint64_t>(), true);
180 else
181 field->store(value.GetValue<int64_t>(), false);
182 break;
183 }
184 case MYSQL_TYPE_FLOAT:
185 field->store(value.GetValue<float>());
186 break;
187 case MYSQL_TYPE_DOUBLE:
188 field->store(value.GetValue<double>());
189 break;
190 case MYSQL_TYPE_DATE:
191 case MYSQL_TYPE_NEWDATE:
192 case MYSQL_TYPE_DATETIME:
193 case MYSQL_TYPE_DATETIME2:
194 case MYSQL_TYPE_TIMESTAMP:
195 case MYSQL_TYPE_TIMESTAMP2: {
196 auto str = value.GetValue<duckdb::string>();
197 MYSQL_TIME tm;
198 MYSQL_TIME_STATUS status;
199 my_time_status_init(&status);
200 str_to_datetime_or_date(str.c_str(), str.size(), &tm, 0, &status);
201 store_field_temporal_value(field, &tm);
202 break;
203 }
204 case MYSQL_TYPE_TIME:
205 case MYSQL_TYPE_TIME2: {
206 auto str = value.GetValue<duckdb::string>();
207 MYSQL_TIME tm;
208 MYSQL_TIME_STATUS status;
209 my_time_status_init(&status);
210 str_to_DDhhmmssff(str.c_str(), str.size(), &tm, TIME_MAX_HOUR, &status);
211 store_field_temporal_value(field, &tm);
212 break;
213 }
214 default: {
215 auto str = value.GetValue<duckdb::string>();
216 field->store(str.c_str(), str.size(), system_charset_info);
217 break;
218 }
219 }

223{

226 if (i != 0)

251bool can_pushdown_to_parquet(SELECT_LEX *sel_lex,
252 std::vector<TABLE_LIST *> &parquet_tables,
253 std::vector<TABLE_LIST *> &external_tables)
254{
255 std::unordered_set<std::string> seen_parquet_names;
256 std::unordered_set<std::string> seen_external_names;
257 bool has_parquet_table = false;
258
259 for (TABLE_LIST *tbl = sel_lex->get_table_list(); tbl; tbl = tbl->next_global) {
260 if (tbl->derived || !tbl->table || !tbl->table->file)
261 return false;
262
263 const std::string table_name(tbl->table_name.str);
264 if (tbl->table->file->ht == parquet_hton) {
265 has_parquet_table = true;
266 if (seen_parquet_names.insert(table_name).second)
267 parquet_tables.push_back(tbl);
268 } else if (seen_external_names.insert(table_name).second) {
269 external_tables.push_back(tbl);
270 }
271 }
272
273 return has_parquet_table;
274}

276void register_external_table_names(TABLE_LIST *tbl)
277{
278 if (!tbl || !tbl->table)
279 return;
280
281 std::unordered_set<std::string> registered_names;
282 auto register_name = [&](const char *name) {
283 if (!name || !name[0])
284 return;
285
286 const std::string key(name);
287 if (registered_names.insert(key).second)
288 myparquet::register_external_table(key, tbl->table);
289 };
290
291 register_name(tbl->table_name.str);
292 register_name(tbl->alias.str);
293
294 if (tbl->db.str && tbl->db.str[0]) {
295 const std::string qualified_name =
296 std::string(tbl->db.str) + "." + tbl->table_name.str;
297 if (registered_names.insert(qualified_name).second)
298 myparquet::register_external_table(qualified_name, tbl->table);
299
300 if (tbl->alias.str && std::strcmp(tbl->alias.str, tbl->table_name.str) != 0) {
301 const std::string qualified_alias =
302 std::string(tbl->db.str) + "." + tbl->alias.str;
303 if (registered_names.insert(qualified_alias).second)
304 myparquet::register_external_table(qualified_alias, tbl->table);
305 }
306 }
307}

336int ha_parquet_select_handler::init_scan()
337{
338 DBUG_ENTER("ha_parquet_select_handler::init_scan");
339
340 duckdb_db = std::make_unique<duckdb::DuckDB>(nullptr);
341 myparquet::register_cross_engine_scan(*duckdb_db->instance);
342 duckdb_con = std::make_unique<duckdb::Connection>(*duckdb_db);
343 parquet_log_info("DuckDB select handler initialized");
344 query_result.reset();
345 current_chunk.reset();
346 current_row_index = 0;
347
348 auto cleanup_external_registry = [&]() {
349 if (has_cross_engine)
350 myparquet::clear_external_tables();
351 };
352
353 parquet_log_info("DuckDB query [select/install-parquet] INSTALL parquet;");
354 duckdb_con->Query("INSTALL parquet;");
355 parquet_log_info("DuckDB query [select/load-parquet] LOAD parquet;");
356 duckdb_con->Query("LOAD parquet;");
357 parquet_log_info("DuckDB query [select/install-httpfs] INSTALL httpfs;");
358 duckdb_con->Query("INSTALL httpfs;");
359 parquet_log_info("DuckDB query [select/load-httpfs] LOAD httpfs;");
360 duckdb_con->Query("LOAD httpfs;");
361
362 parquet::ObjectStoreConfig scan_object_store_config;
363 bool have_scan_object_store_config = false;
364
365 for (TABLE_LIST *tbl : parquet_tables) {
366 parquet::TableMetadata metadata;
367 std::string error_message;
368 if (!parquet::ResolveRuntimeTableMetadata(
369 tbl->table->s->normalized_path.str, &metadata, &error_message) ||
370 !parquet::ValidateCatalogConfig(metadata, &error_message) ||
371 !parquet::ValidateObjectStoreConfig(metadata, true, &error_message)) {
372 cleanup_external_registry();
373 my_error(ER_UNKNOWN_ERROR, MYF(0), error_message.c_str());
374 DBUG_RETURN(HA_ERR_INTERNAL_ERROR);
375 }
376
377 if (!have_scan_object_store_config) {
378 scan_object_store_config = metadata.object_store_config;
379 have_scan_object_store_config = true;
380 } else if (!duckdb_configs_compatible(scan_object_store_config,
381 metadata.object_store_config)) {
382 cleanup_external_registry();
383 my_error(ER_UNKNOWN_ERROR, MYF(0),
384 "Parquet pushdown currently requires matching object-store "
385 "credentials and endpoint settings across Parquet tables");
386 DBUG_RETURN(HA_ERR_INTERNAL_ERROR);
387 }
388
389 if (!configure_duckdb_s3(duckdb_con.get(), metadata.object_store_config,
390 &error_message)) {
391 cleanup_external_registry();
392 my_error(ER_UNKNOWN_ERROR, MYF(0), error_message.c_str());
393 DBUG_RETURN(HA_ERR_INTERNAL_ERROR);
394 }
395
396 std::vector<std::string> s3_files;
397 long http_code = 0;
398 if (!resolve_parquet_data_files(metadata, &s3_files, &http_code)) {
399 cleanup_external_registry();
400 my_error(ER_UNKNOWN_ERROR, MYF(0),
401 "Failed to fetch LakeKeeper metadata for Parquet pushdown");
402 DBUG_RETURN(HA_ERR_INTERNAL_ERROR);
403 }
404 if (http_code != 200) {
405 cleanup_external_registry();
406 my_error(ER_UNKNOWN_ERROR, MYF(0),
407 "LakeKeeper returned a non-200 response for Parquet pushdown");
408 DBUG_RETURN(HA_ERR_INTERNAL_ERROR);
409 }
410
411 const std::string create_view_sql = s3_files.empty()
412 ? build_empty_view_query(tbl)
413 : build_read_view_query(tbl, s3_files);
414 if (create_view_sql.empty()) {
415 cleanup_external_registry();
416 my_error(ER_UNKNOWN_ERROR, MYF(0),
417 "Parquet pushdown could not map the table schema to DuckDB");
418 DBUG_RETURN(HA_ERR_INTERNAL_ERROR);
419 }
420
421 parquet_log_info("DuckDB query [select/create-parquet-view] " +
422 parquet_log_preview(create_view_sql));
423 auto create_view_result = duckdb_con->Query(create_view_sql);
424 if (!create_view_result || create_view_result->HasError()) {
425 cleanup_external_registry();
426 const std::string error_message =
427 create_view_result ? create_view_result->GetError()
428 : "DuckDB returned a null result while creating a Parquet view";
429 my_error(ER_UNKNOWN_ERROR, MYF(0), error_message.c_str());
430 DBUG_RETURN(HA_ERR_INTERNAL_ERROR);
431 }
432 }
433
434 if (have_scan_object_store_config) {
435 std::string error_message;
436 if (!configure_duckdb_s3(duckdb_con.get(), scan_object_store_config,
437 &error_message)) {
438 cleanup_external_registry();
439 my_error(ER_UNKNOWN_ERROR, MYF(0), error_message.c_str());
440 DBUG_RETURN(HA_ERR_INTERNAL_ERROR);
441 }
442 }
443
444 if (has_cross_engine) {
445 for (TABLE_LIST *tbl : external_tables)
446 register_external_table_names(tbl);
447
448 const std::string parquet_desc = describe_tables(parquet_tables);
449 const std::string external_desc = describe_tables(external_tables);
450 sql_print_information(
451 "Parquet: cross-engine pushdown init parquet_tables=[%s] external_tables=[%s]",
452 parquet_desc.c_str(), external_desc.c_str());
453 }
454
455 std::string sql(query_string.ptr(), query_string.length());
456 parquet_log_info("DuckDB query [select/execute] " +
457 parquet_log_preview(sql));
458 query_result = duckdb_con->Query(sql);
459
460 if (!query_result || query_result->HasError()) {
461 cleanup_external_registry();
462 const std::string error_message =
463 query_result ? query_result->GetError()
464 : "DuckDB returned a null result for Parquet pushdown";
465 my_error(ER_UNKNOWN_ERROR, MYF(0), error_message.c_str());
466 DBUG_RETURN(HA_ERR_INTERNAL_ERROR);
467 }
468
469 DBUG_RETURN(0);
470}

472int ha_parquet_select_handler::next_row()
473{
474 DBUG_ENTER("ha_parquet_select_handler::next_row");
475
476 if (!query_result)
477 DBUG_RETURN(HA_ERR_INTERNAL_ERROR);
478
479 if (!current_chunk || current_row_index >= current_chunk->size()) {
480 current_chunk.reset();
481 current_chunk = query_result->Fetch();
482
483 if (!current_chunk || current_chunk->size() == 0)
484 DBUG_RETURN(HA_ERR_END_OF_FILE);
485
486 current_row_index = 0;
487 }
488
489 size_t col_count = current_chunk->ColumnCount();
490 size_t field_count = 0;
491 for (Field **f = table->field; *f; ++f)
492 field_count++;
493
494 size_t ncols = (col_count < field_count) ? col_count : field_count;
495 for (size_t col_idx = 0; col_idx < ncols; ++col_idx) {
496 duckdb::Value value = current_chunk->GetValue(col_idx, current_row_index);
497 Field *field = table->field[col_idx];
498 store_duckdb_field_in_mysql_format(field, value, thd);
499 }
500
501 current_row_index++;
502 DBUG_RETURN(0);
503}
